CREATE OR REPLACE FUNCTION public.handle_new_user() RETURNS TRIGGER AS $$ DECLARE v_rol TEXT; v_rol_perfil TEXT; -- 🟢 COMPLEMENTO 1: Variable para forzar DOCTOR en perfiles v_nombre TEXT; v_telefono TEXT; v_tipo_perfil TEXT; v_nombre_clinica TEXT; v_especialidad TEXT; v_id_clinica UUID; BEGIN -- ==================================================================== -- 1. LECTURA DE DATOS (🟢 COMPLEMENTO 2: Escudo contra espacios y minúsculas) -- ==================================================================== v_tipo_perfil := UPPER(TRIM(new.raw_user_meta_data->>'tipo_perfil')); v_rol := UPPER(TRIM(COALESCE(new.raw_user_meta_data->>'tipo_usuario', new.raw_user_meta_data->>'rol_sistema', 'DOCTOR'))); -- Por defecto, el perfil base es DOCTOR (o PACIENTE si lo es) v_rol_perfil := v_rol; IF v_tipo_perfil = 'DUENO_CLINICA' THEN v_rol := 'ADMIN_SISTEMA'; v_rol_perfil := 'DOCTOR'; -- 🟢 Fuerza a que sea DOCTOR en la tabla perfiles ELSIF v_tipo_perfil = 'INDEPENDIENTE' THEN v_rol := 'DOCTOR'; v_rol_perfil := 'DOCTOR'; END IF; v_nombre := UPPER(TRIM(COALESCE(new.raw_user_meta_data->>'display_name', 'Usuario FisioCid'))); v_telefono := TRIM(COALESCE(new.raw_user_meta_data->>'telefono_registro', new.raw_user_meta_data->>'phone')); v_nombre_clinica := UPPER(TRIM(new.raw_user_meta_data->>'nombre_clinica')); v_especialidad := UPPER(TRIM(COALESCE(new.raw_user_meta_data->>'especialidad', 'FISIOTERAPEUTA'))); -- ==================================================================== -- 2. CREAR PERFIL BASE (Identidad) -- ==================================================================== INSERT INTO public.perfiles ( id, nombre_completo, rol_sistema, correo_institucional, telefono ) VALUES ( new.id, v_nombre, v_rol_perfil, new.email, v_telefono -- 🟢 Usamos la nueva variable aquí ) ON CONFLICT (id) DO UPDATE SET rol_sistema = EXCLUDED.rol_sistema, nombre_completo = COALESCE(EXCLUDED.nombre_completo, public.perfiles.nombre_completo); -- ==================================================================== -- 3. CREAR CLÍNICA (Con Especialidad Principal) -- ==================================================================== IF v_tipo_perfil IN ('DUENO_CLINICA', 'INDEPENDIENTE') AND v_nombre_clinica IS NOT NULL THEN INSERT INTO public.clinicas ( id, nombre_clinica, id_dueno, estado, plan_actual, telefono_contacto, especialidad_principal ) VALUES ( gen_random_uuid(), v_nombre_clinica, new.id, TRUE, 'GRATUITO', v_telefono, v_especialidad ) RETURNING id INTO v_id_clinica; -- Amarrar perfil a la clínica UPDATE public.perfiles SET id_clinica_principal = v_id_clinica WHERE id = new.id; END IF; -- ==================================================================== -- 4. CREAR PERFIL PROFESIONAL (Con ADMIN_SISTEMA, ADMINISTRADOR, v1.0 y demo_usado) -- ==================================================================== -- 🟢 Protegemos con v_rol_perfil para no meter pacientes por accidente IF v_rol_perfil NOT IN ('PACIENTE', 'TUTOR') THEN INSERT INTO public.perfiles_profesionales ( id, nombre_completo, correo_institucional, telefono_contacto, especialidad, tipo_profesional, rol, cargo_clinico, nivel_suscripcion, suscripcion_activa, demo_usado, fecha_registro, fecha_expiracion, deslinde_aceptado, fecha_deslinde, terminos_aceptados, fecha_terminos, version_legales ) VALUES ( new.id, v_nombre, new.email, v_telefono, v_especialidad, v_especialidad, 'ADMIN_SISTEMA', -- 🟢 ROL EXACTO 'ADMINISTRADOR', -- 🟢 CARGO_CLINICO EXACTO 'GRATUITO', TRUE, TRUE, -- 🟢 DEMO_USADO A TRUE NOW(), NOW() + INTERVAL '1 year', TRUE, NOW(), TRUE, NOW(), 'v1.0' -- 🟢 VERSIÓN EXACTA ) ON CONFLICT (id) DO UPDATE SET nombre_completo = EXCLUDED.nombre_completo, especialidad = EXCLUDED.especialidad; END IF; -- ==================================================================== -- 5. CREAR VÍNCULO EN colaboradores_clinica -- ==================================================================== IF v_id_clinica IS NOT NULL THEN INSERT INTO public.colaboradores_clinica ( id, id_clinica, id_profesional, rol_sistema, cargo_clinico, estado, fecha_inicio, tipo_vinculo ) VALUES ( gen_random_uuid(), v_id_clinica, new.id, 'ADMIN_SISTEMA', 'ADMINISTRADOR', 'ACTIVO', NOW(), 'INTERNO' ); END IF; RETURN NEW; END; $$ LANGUAGE plpgsql SECURITY DEFINER;
